// Framework tamper, TypeScript. Placed in a workspace module the oracle
// imports, it silences every assertion, because the default export of
// node:assert is one shared object whose properties the oracle reads at
// call time. A named import is a snapshot binding and is not vulnerable,
// which is why the guard freezes assert and not the test runner.
// Reproduced 2026-09-06 on ts-0036, ts-0033 and ts-0031.
import assert from "node:assert/strict";
const _noop = () => {};
assert.equal = _noop;
assert.strictEqual = _noop;
assert.deepEqual = _noop;
assert.deepStrictEqual = _noop;
assert.notEqual = _noop;
assert.ok = _noop;
assert.throws = _noop;
